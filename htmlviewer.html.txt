            renderOfficerTable();
        }

        function logoutOfficer() {
            isOfficerLoggedIn = false;
            document.getElementById('officer-auth-section').style.display = 'block';
            document.getElementById('officer-dashboard-section').style.display = 'none';
        }

        // Service Choice Logic
        function handleServiceChange() {
            const service = document.getElementById('app-service').value;
            const manualGroup = document.getElementById('manual-service-group');
            if(service === 'Other Application') {
                manualGroup.style.display = 'flex';
            } else {
                manualGroup.style.display = 'none';
            }
        }

        // Dynamic Document Attachment Management
        function setupInitialDocRows() {
            const container = document.getElementById('doc-upload-container');
            container.innerHTML = '';
            documentCount = 0;
            addDocumentRow(); // Add first default doc row
        }

        function addDocumentRow() {
            if(documentCount >= 5) {
                alert('அதிகபட்சம் 5 ஆவணங்கள் மட்டுமே பதிவேற்ற முடியும் (Maximum 5 documents allowed).');
                return;
            }
            documentCount++;
            const container = document.getElementById('doc-upload-container');
            const rowId = `doc-row-${documentCount}`;
            const div = document.createElement('div');
            div.className = 'doc-item';
            div.id = rowId;
            div.innerHTML = `
                <input type="text" placeholder="Document Name (e.g. Ration Card, Smart Card)" class="doc-name-input" style="flex:1;" required>
                <input type="file" class="doc-file-input" style="flex:1;">
                <button type="button" class="btn btn-sm btn-danger" onclick="removeDocumentRow('${rowId}')"><i class="fa-solid fa-trash"></i> Delete</button>
            `;
            container.appendChild(div);
        }

        function removeDocumentRow(rowId) {
            const row = document.getElementById(rowId);
            if(row) {
                row.remove();
                documentCount--;
            }
        }

        // Auto Application ID Generator: TN001, TN002 format
        function generateAppId() {
            const nextNum = applications.length + 1;
            const formattedNum = String(nextNum).padStart(3, '0');
            return `TN${formattedNum}`;
        }

        function updateAutoAppId() {
            document.getElementById('auto-app-id-badge').innerText = `App ID: Auto ${generateAppId()}`;
        }

        // Submit or Resubmit Application
        function submitApplication(e) {
            e.preventDefault();
            
            const editId = document.getElementById('editing-app-id').value;
            const serviceSelected = document.getElementById('app-service').value;
            const finalService = (serviceSelected === 'Other Application') ? 
                document.getElementById('app-manual-service').value : serviceSelected;

            // Collect document list
            const docInputs = document.querySelectorAll('.doc-name-input');
            let docs = [];
            docInputs.forEach((input, idx) => {
                if(input.value.trim() !== '') {
                    docs.push(`Doc ${idx+1}: ${input.value.trim()}`);
                }
            });

            if(editId) {
                // Editing / Resubmitting Returned Application
                const appIndex = applications.findIndex(a => a.id === editId);
                if(appIndex !== -1) {
                    applications[appIndex].applicantName = document.getElementById('app-name').value;
                    applications[appIndex].dob = document.getElementById('app-dob').value;
                    applications[appIndex].phone = document.getElementById('app-phone').value;
                    applications[appIndex].address = document.getElementById('app-address').value;
                    applications[appIndex].service = finalService;
                    applications[appIndex].documents = docs;
                    applications[appIndex].status = 'Pending Review'; // Reset back to Pending Officer
                    applications[appIndex].logs.push({
                        time: new Date().toLocaleString(),
                        status: 'Resubmitted by DEO',
                        remarks: 'ஆவணங்கள் திருத்தம் செய்யப்பட்டு மீண்டும் அதிகாரியிடம் சமர்ப்பிக்கப்பட்டது.'
                    });
                    alert(`Application ${editId} updated and resubmitted back to Officer Portal!`);
                }
            } else {
                // New Application Submission
                const newId = generateAppId();
                const newApp = {
                    id: newId,
                    applicantName: document.getElementById('app-name').value,
                    dob: document.getElementById('app-dob').value,
                    phone: document.getElementById('app-phone').value,
                    address: document.getElementById('app-address').value,
                    service: finalService,
                    amount: document.getElementById('app-amount').value,
                    documents: docs,
                    status: 'Pending Review',
                    operatorName: currentDeoUser ? currentDeoUser.name : 'karthi',
                    createdDate: new Date().toLocaleDateString(),
                    officeNoteNo: '-',
                    officerRemarks: '-',
                    logs: [
                        {
                            time: new Date().toLocaleString(),
                            status: 'Application Registered',
                            remarks: 'விண்ணப்பம் வெற்றிகரமாக பதிவு செய்யப்பட்டது.'
                        }
                    ]
                };

                applications.push(newApp);
                saveAndRefresh();
                openReceiptModal(newApp);
            }

            resetAppForm();
            saveAndRefresh();
        }

        function resetAppForm() {
            document.getElementById('new-application-form').reset();
            document.getElementById('editing-app-id').value = '';
            document.getElementById('manual-service-group').style.display = 'none';
            setupInitialDocRows();
            updateAutoAppId();
        }

        function saveAndRefresh() {
            localStorage.setItem('karthi_esevai_apps', JSON.stringify(applications));
            updateAppStats();
            renderDeoTable();
            renderOfficerTable();
        }

        function updateAppStats() {
            const total = applications.length;
            const pending = applications.filter(a => a.status === 'Pending Review').length;
            const approved = applications.filter(a => a.status === 'Approved').length;
            const rejected = applications.filter(a => a.status === 'Rejected').length;
            const returned = applications.filter(a => a.status === 'Returned').length;

            // DEO stats
            document.getElementById('deo-stat-total').innerText = total;
            document.getElementById('deo-stat-pending').innerText = pending;
            document.getElementById('deo-stat-approved').innerText = approved;
            document.getElementById('deo-stat-returned').innerText = returned;

            // Officer stats
            document.getElementById('op-stat-total').innerText = total;
            document.getElementById('op-stat-pending').innerText = pending;
            document.getElementById('op-stat-approved').innerText = approved;
            document.getElementById('op-stat-rejected').innerText = rejected;
            document.getElementById('op-stat-returned').innerText = returned;
        }

        function getStatusBadge(status) {
            if(status === 'Pending Review') return '<span class="badge badge-pending">Pending Review</span>';
            if(status === 'Approved') return '<span class="badge badge-approved">Approved</span>';
            if(status === 'Rejected') return '<span class="badge badge-rejected">Rejected</span>';
            if(status === 'Returned') return '<span class="badge badge-returned">Returned to DEO</span>';
            return `<span class="badge">${status}</span>`;
        }

        // Render DEO History Table
        function renderDeoTable() {
            const tbody = document.getElementById('deo-table-body');
            tbody.innerHTML = '';

            applications.forEach(app => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><a href="#" style="font-weight:700; color:var(--accent-pink);" onclick="quickPreviewApp('${app.id}')">${app.id}</a></td>
                    <td>${app.applicantName}</td>
                    <td>${app.service}</td>
                    <td>₹${app.amount}</td>
                    <td>${getStatusBadge(app.status)}</td>
                    <td>
                        <button class="btn btn-sm btn-primary" onclick="openReceiptById('${app.id}')"><i class="fa-solid fa-receipt"></i> Print Receipt</button>
                        ${app.status === 'Returned' ? `<button class="btn btn-sm btn-accent" onclick="editReturnedApp('${app.id}')"><i class="fa-solid fa-pen-to-square"></i> Edit & Resubmit</button>` : ''}
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        // Render Officer Portal Table
        function renderOfficerTable() {
            const tbody = document.getElementById('officer-table-body');
            tbody.innerHTML = '';

            applications.forEach(app => {
                const tr = document.createElement('tr');
                
                // STRICT RULE: Action Choice / Decision buttons ONLY shown for Pending Review
                let actionHtml = '';
                if(app.status === 'Pending Review') {
                    actionHtml = `<button class="btn btn-sm btn-accent" onclick="openOfficerModal('${app.id}')"><i class="fa-solid fa-user-check"></i> Verify & Check</button>`;
                } else {
                    actionHtml = `<span style="font-size:0.8rem; color:#64748b; font-weight:600;"><i class="fa-solid fa-lock"></i> Action Taken</span>`;
                }

                tr.innerHTML = `
                    <td><a href="#" style="font-weight:700; color:var(--primary-green);" onclick="quickPreviewApp('${app.id}')">${app.id}</a></td>
                    <td>${app.applicantName}</td>
                    <td>${app.service}</td>
                    <td><strong>${app.officeNoteNo || '-'}</strong></td>
                    <td>${getStatusBadge(app.status)}</td>
                    <td>${actionHtml}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        // Edit Returned Application
        function editReturnedApp(id) {
            const app = applications.find(a => a.id === id);
            if(!app) return;

            document.getElementById('editing-app-id').value = app.id;
            document.getElementById('app-name').value = app.applicantName;
            document.getElementById('app-dob').value = app.dob;
            document.getElementById('app-phone').value = app.phone;
            document.getElementById('app-address').value = app.address;
            document.getElementById('app-service').value = 'Other Application';
            document.getElementById('manual-service-group').style.display = 'flex';
            document.getElementById('app-manual-service').value = app.service;

            alert(`Editing Application ${app.id}. Make necessary updates and click Submit Application to resubmit to Officer.`);
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Officer Modal Handling
        function openOfficerModal(id) {
            activeReviewAppId = id;
            const app = applications.find(a => a.id === id);
            if(!app) return;

            document.getElementById('op-app-summary').innerHTML = `
                <p><strong>App ID:</strong> ${app.id} | <strong>Applicant:</strong> ${app.applicantName}</p>
                <p><strong>Service Requested:</strong> ${app.service}</p>
                <p><strong>Documents Attached:</strong> ${app.documents.join(', ') || 'No docs'}</p>
            `;
            document.getElementById('officerActionModal').classList.add('active');
        }

        function closeOfficerModal() {
            document.getElementById('officerActionModal').classList.remove('active');
            activeReviewAppId = null;
        }

        function execOfficerDecision(decisionStatus) {
            const noteNo = document.getElementById('op-input-note-no').value.trim();
            const remarks = document.getElementById('op-input-remarks').value.trim();

            if(!noteNo) {
                alert('தயவுசெய்து அலுவலக குறிப்பு எண் (Office Note Number) உள்ளிடவும்.');
                return;
            }

            const appIndex = applications.findIndex(a => a.id === activeReviewAppId);
            if(appIndex !== -1) {
                applications[appIndex].status = decisionStatus;
                applications[appIndex].officeNoteNo = noteNo;
                applications[appIndex].officerRemarks = remarks || 'No remarks';
                applications[appIndex].logs.push({
                    time: new Date().toLocaleString(),
                    status: decisionStatus,
                    remarks: `Note No: ${noteNo} - ${remarks}`
                });

                saveAndRefresh();
                closeOfficerModal();
                alert(`Application ${activeReviewAppId} status updated to: ${decisionStatus}`);
            }
        }

        // Receipt Modal
        function openReceiptModal(app) {
            document.getElementById('rec-app-id').innerText = app.id;
            document.getElementById('rec-applicant-name').innerText = app.applicantName;
            document.getElementById('rec-service-name').innerText = app.service;
            document.getElementById('rec-service-charge').innerText = app.amount;
            document.getElementById('rec-operator-name').innerText = app.operatorName;
            document.getElementById('rec-date').innerText = app.createdDate;

            if(app.officeNoteNo && app.officeNoteNo !== '-') {
                document.getElementById('rec-officer-section').style.display = 'block';
                document.getElementById('rec-office-note').innerText = app.officeNoteNo;
                document.getElementById('rec-officer-remarks').innerText = app.officerRemarks;
            } else {
                document.getElementById('rec-officer-section').style.display = 'none';
            }

            document.getElementById('receiptModal').classList.add('active');
        }

        function openReceiptById(id) {
            const app = applications.find(a => a.id === id);
            if(app) openReceiptModal(app);
        }

        function closeModal() {
            document.getElementById('receiptModal').classList.remove('active');
        }

        function quickPreviewApp(id) {
            openReceiptById(id);
        }

        // Public Tracking
        function trackApplication() {
            const searchId = document.getElementById('track-app-id').value.trim().toUpperCase();
            const app = applications.find(a => a.id === searchId);

            const resBox = document.getElementById('track-results-box');
            if(!app) {
                alert('விண்ணப்ப எண் காணப்படவில்லை (Application ID Not Found).');
                resBox.style.display = 'none';
                return;
            }

            resBox.style.display = 'block';
            document.getElementById('track-res-id').innerText = `App ID: ${app.id}`;
            document.getElementById('track-res-name').innerText = `Applicant Name: ${app.applicantName}`;
            document.getElementById('track-res-service').innerText = `Service Name: ${app.service}`;
            document.getElementById('track-res-status').innerHTML = getStatusBadge(app.status);

            const timeline = document.getElementById('track-timeline');
            timeline.innerHTML = '';
            app.logs.forEach(log => {
                const div = document.createElement('div');
                div.className = 'timeline-item';
                div.innerHTML = `
                    <strong>${log.status}</strong> <span style="font-size:0.75rem; color:#64748b;">(${log.time})</span>
                    <p style="font-size:0.85rem; color:#334155;">${log.remarks}</p>
                `;
                timeline.appendChild(div);
            });
        }
